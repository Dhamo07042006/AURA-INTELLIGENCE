----------------------------------------------------------------------------------------------------

to send HTTP POST requests with the following JSON schema:

Parameter Name    | Data Type | Accepted Values / Format                 | ML Feature Mapping
------------------+-----------+------------------------------------------+--------------------------------------
hospital_id       | string    | "demo-hospital"                          | Multi-tenant hospital scoping
device_id         | string    | e.g. "DEV000001" (Range: DEV000001–DEV009434)| Device registry & historical lookup
device_type       | string    | "Ventilator", "Patient monitor",         | Component ontology & baseline
                  |           | "CT scanner", "MRI scanner",             |
                  |           | "Infusion pump", "Defibrillator", etc.   |
department        | string    | "Intensive Care Unit (ICU)",             | Department risk aggregation
                  |           | "Radiology Department",                  |
                  |           | "Clinical Laboratory",                   |
                  |           | "Operating Rooms (OR)", "General Ward"   |
timestamp         | string    | ISO-8601 UTC (e.g. "2026-08-16T15:30:00Z")| Time-series feature calculation
battery_health    | float     | 0.0 to 100.0 (%)                         | Approx_Battery_Health feature
temperature       | float     | 30.0 to 65.0 (°C)                        | Average_Temperature_C feature
load_percent      | float     | 0.0 to 100.0 (%)                         | Peak_Load_Percent feature
error_code        | string    | "OK", "BAT_CRITICAL", "TEMP_CRITICAL",   | Error frequency feature injection
                  |           | "SENSOR_ERR", "POWER_FLUC", "SYS_RESET"  |
operating_hours   | float     | e.g. 1200.5 (hours)                      | Approx_Operating_Hours feature
voltage           | float     | 10.0 to 28.0 (V)                         | Voltage_Fluctuation_Count feature
humidity          | float     | 20.0 to 90.0 (%) (Optional)              | Average_Humidity_Percent feature
battery_cycles    | integer   | e.g. 240 (Optional)                      | Approx_Battery_Cycles feature
days_since_maint  | integer   | e.g. 30 (Optional)                       | Days_Since_Last_Maintenance feature


3. ERROR CODES — REAL-TIME ML FEATURE INJECTION RULES
----------------------------------------------------------------------------------------------------
ERROR CODE      | ML FEATURES INJECTED BY AURA BACKEND             | PREDICTED RISK LEVEL RESULT
----------------+--------------------------------------------------+-----------------------------------
"OK"            | Nominal baseline — uses raw battery/temp/load    | LOW Risk (No notification)
"BAT_WARN"      | Battery_Errors_Last_7_Days += 1                  | MEDIUM Warning
"BAT_CRITICAL"  | Battery_Errors_Last_7_Days += 10, Alarms += 5    | 🚨 CRITICAL (Triggers Inbox Alert)
"TEMP_WARN"     | Alarms_Last_7_Days += 3                          | MEDIUM / HIGH Warning
"TEMP_CRITICAL" | Alarms_Last_7_Days += 5                          | 🚨 CRITICAL (Triggers Inbox Alert)
"SENSOR_ERR"    | Sensor_Errors_Last_7_Days += 10, Abnormal += 10 | 🚨 HIGH / CRITICAL
"POWER_FLUC"    | Power_Errors_Last_7_Days += 8, VoltageFluc += 12 | 🚨 HIGH / CRITICAL
"SYS_RESET"     | System_Resets_Last_7_Days += 5                   | 🚨 HIGH / CRITICAL

    <div class="card">

    <script>
        let timer = null;

        function log(msg, riskClass = 'log-low') {
            const c = document.getElementById('console');
            const time = new Date().toLocaleTimeString();
            c.innerHTML += `<div class="${riskClass}">[${time}] ${msg}</div>`;
            c.scrollTop = c.scrollHeight;
        }

        function getPayload(errorCode = null) {
            const sc = errorCode || document.getElementById('scenario').value;
            const devId = document.getElementById('deviceId').value;
            const devType = document.getElementById('deviceType').value;
            
            let bat = 92.5, temp = 36.8, load = 62.0, volt = 23.8;
            if (sc === 'BAT_CRITICAL') { bat = 14.2; temp = 37.4; load = 88.0; volt = 18.5; }
            else if (sc === 'TEMP_CRITICAL') { bat = 75.0; temp = 54.2; load = 96.0; volt = 21.0; }
            else if (sc === 'SENSOR_ERR') { bat = 82.0; temp = 39.1; load = 78.0; volt = 22.5; }
            else if (sc === 'POWER_FLUC') { bat = 68.0; temp = 41.0; load = 91.0; volt = 16.2; }

            return {
                hospital_id: "demo-hospital",
                device_id: devId,
                device_type: devType,
                department: "Intensive Care Unit (ICU)",
                timestamp: new Date().toISOString(),
                battery_health: bat,
                temperature: temp,
                load_percent: load,
                error_code: sc,
                operating_hours: 3400.0,
                voltage: volt
            };
        }

        async function sendSinglePayload(forceCode = null) {
            const ip = document.getElementById('serverIp').value;
            const port = document.getElementById('serverPort').value;
            const url = `http://${ip}:${port}/api/v1/ingest/telemetry`;
            const payload = getPayload(forceCode);

            try {
                const res = await fetch(url, {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'X-API-Key': 'aura_live_ingest_key_2026',
                        'Hospital-ID': 'demo-hospital'
                    },
                    body: JSON.stringify(payload)
                });
                const data = await res.json();
                const risk = data.risk_level || 'UNKNOWN';
                const cls = risk === 'CRITICAL' ? 'log-critical' : (risk === 'HIGH' ? 'log-high' : 'log-low');
                log(`POST ${payload.device_id} | Code: ${payload.error_code} | Bat: ${payload.battery_health}% | Temp: ${payload.temperature}°C ➔ ML Prediction: ${risk} (Health: ${data.overall_health}%, Anomaly: ${data.anomaly_score})`, cls);
            } catch (err) {
                log(`❌ Transmission Error: ${err.message}`, 'log-critical');
            }
        }

        function startStreaming() {
            document.getElementById('startBtn').style.display = 'none';
            document.getElementById('stopBtn').style.display = 'block';
            const interval = parseInt(document.getElementById('interval').value);
            log('Starting live telemetry stream to AURA ML engine...');
            sendSinglePayload();
            timer = setInterval(() => sendSinglePayload(), interval);
        }

        function stopStreaming() {
            clearInterval(timer);
            document.getElementById('startBtn').style.display = 'block';
            document.getElementById('stopBtn').style.display = 'none';
            log('Telemetry stream stopped.');
        }
    </script>
</body>
</html>
